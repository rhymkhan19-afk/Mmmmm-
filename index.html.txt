        let prayerTimesData = {};
        let countdownInterval;

        function fetchPrayerTimes() {
            if (navigator.geolocation) {
                document.getElementById('prayer-loading').style.display = 'block';
                navigator.geolocation.getCurrentPosition(
                    position => getAladhanAPI(position.coords.latitude, position.coords.longitude),
                    error => alert("يرجى السماح بتحديد الموقع لعرض الأوقات.")
                );
            }
        }

        function getAladhanAPI(lat, lon) {
            fetch(`https://api.aladhan.com/v1/timings?latitude=${lat}&longitude=${lon}&method=4`)
                .then(res => res.json())
                .then(data => {
                    prayerTimesData = data.data.timings;
                    renderPrayerGrid(prayerTimesData);
                    startCountdown();
                    document.getElementById('prayer-loading').style.display = 'none';
                    document.getElementById('prayer-grid').style.display = 'grid';
                });
        }

        function renderPrayerGrid(times) {
            const prayers = [
                { name: 'الفجر', id: 'Fajr' },
                { name: 'الشروق', id: 'Sunrise' },
                { name: 'الظهر', id: 'Dhuhr' },
                { name: 'العصر', id: 'Asr' },
                { name: 'المغرب', id: 'Maghrib' },
                { name: 'العشاء', id: 'Isha' }
            ];
            
            let html = '';
            prayers.forEach(p => {
                html += `
                    <div class="prayer-card" id="card-${p.id}">
                        <div class="prayer-name">${p.name}</div>
                        <div class="prayer-time">${times[p.id]}</div>
                    </div>
                `;
            });
            document.getElementById('prayer-grid').innerHTML = html;
        }

        function startCountdown() {
            if(countdownInterval) clearInterval(countdownInterval);
            
            countdownInterval = setInterval(() => {
                const now = new Date();
                const currentTime = now.getHours() * 60 + now.getMinutes();
                
                const prayers = [
                    { name: 'الفجر', id: 'Fajr' },
                    { name: 'الظهر', id: 'Dhuhr' },
                    { name: 'العصر', id: 'Asr' },
                    { name: 'المغرب', id: 'Maghrib' },
                    { name: 'العشاء', id: 'Isha' }
                ];

                let nextPrayer = null;
                let nextPrayerTimeMinutes = 0;

                // إعادة تعيين الألوان
                document.querySelectorAll('.prayer-card').forEach(c => c.classList.remove('active'));

                for (let p of prayers) {
                    let [hours, minutes] = prayerTimesData[p.id].split(':').map(Number);
                    let timeInMinutes = hours * 60 + minutes;
                    
                    if (timeInMinutes > currentTime) {
                        nextPrayer = p;
                        nextPrayerTimeMinutes = timeInMinutes;
                        break;
                    }
                }

                // إذا مرت العشاء، الصلاة القادمة هي الفجر اليوم التالي
                if (!nextPrayer) {
                    nextPrayer = prayers[0];
                    let [hours, minutes] = prayerTimesData.Fajr.split(':').map(Number);
                    nextPrayerTimeMinutes = (24 * 60) + (hours * 60 + minutes);
                }

                // تفعيل كرت الصلاة القادمة
                const activeCard = document.getElementById(`card-${nextPrayer.id}`);
                if(activeCard) activeCard.classList.add('active');

                // حساب الفرق
                let diff = nextPrayerTimeMinutes - currentTime;
                let h = Math.floor(diff / 60);
                let m = diff % 60;
                let s = 59 - now.getSeconds(); // تقريبي للثواني

                document.getElementById('next-prayer-name').innerText = `باقي على ${nextPrayer.name}`;
                document.getElementById('countdown-timer').innerText = 
                    `${h.toString().padStart(2, '0')}:${m.toString().padStart(2, '0')}:${s.toString().padStart(2, '0')}`;

            }, 1000);
        }
    </script>
